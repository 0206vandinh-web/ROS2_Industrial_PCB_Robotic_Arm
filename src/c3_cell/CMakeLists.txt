cmake_minimum_required(VERSION 3.16)
project(c3_cell LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
foreach(dep rclcpp sensor_msgs geometry_msgs std_msgs rosgraph_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(c3_kin SHARED firmware/kin.c)
target_link_libraries(c3_kin m)
add_library(c3_plant SHARED plugin/plant.cpp firmware/control.c firmware/wire.c)
target_compile_features(c3_plant PRIVATE cxx_std_17)
target_include_directories(c3_plant PRIVATE firmware)
target_link_libraries(c3_plant gz-sim8::gz-sim8 gz-plugin2::register)
ament_target_dependencies(c3_plant rclcpp sensor_msgs geometry_msgs std_msgs rosgraph_msgs)
install(TARGETS c3_kin c3_plant LIBRARY DESTINATION lib)
install(DIRECTORY launch models config DESTINATION share/${PROJECT_NAME})
install(FILES scripts/core.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/station.py scripts/vision.py scripts/gateway.py DESTINATION lib/${PROJECT_NAME})
ament_package()
